## 附录 A：关键 SQL（执行 Agent 直接可用）

```sql
-- A-1 对账四类清单
SELECT d.doc_id, d.status, d.last_indexed, d.department FROM doc_registry d
WHERE d.status='active' AND NOT EXISTS (SELECT 1 FROM rag_vectors v WHERE v.doc_id=d.doc_id);          -- active 无向量
SELECT DISTINCT v.doc_id FROM rag_vectors v
WHERE NOT EXISTS (SELECT 1 FROM doc_registry d WHERE d.doc_id=v.doc_id AND d.status='active');          -- 孤儿向量
SELECT d.doc_id FROM doc_registry d WHERE d.status='active' AND (d.last_indexed IS NULL OR d.last_indexed=''); -- 未标记索引
SELECT v.doc_id, count(*) FROM rag_vectors v JOIN chunk_store c ON c.doc_id=v.doc_id GROUP BY 1
HAVING count(*) <> (SELECT count(*) FROM chunk_store x WHERE x.doc_id=v.doc_id);                        -- 向量/切片数不齐（按需）

-- A-2 覆盖率（K3 验收）
SELECT count(*) active_total,
       count(*) FILTER (WHERE EXISTS (SELECT 1 FROM rag_vectors v WHERE v.doc_id=d.doc_id)) in_vec,
       round(100.0*count(*) FILTER (WHERE EXISTS (SELECT 1 FROM rag_vectors v WHERE v.doc_id=d.doc_id))/count(*),1) pct
FROM doc_registry d WHERE status='active';

-- A-3 parsing 僵尸
SELECT count(*), round(EXTRACT(epoch FROM now()-MIN(updated_at::timestamptz))/3600) oldest_h
FROM doc_registry WHERE status='parsing';

-- A-4 span 闭合抽检（T2.4 验收）
SELECT trace_id, jsonb_array_length(data::jsonb->'spans') span_n, data::jsonb->'spans'
FROM trace_store WHERE created_at::timestamptz>=now()-interval '1 day' AND workflow_name='agent'
ORDER BY created_at DESC LIMIT 10;

-- A-5 Trace 关联率（T2.6 验收）
SELECT round(100.0*count(*) FILTER (WHERE trace_id<>'')/count(*),1) pct
FROM llm_usage WHERE created_at::timestamptz>=now()-interval '7 days';

-- A-6 队列存量（T1.6 验收）
SELECT status, count(*) FROM tasks GROUP BY 1;

-- A-7 TTFT 有效性（T2.3 验收）
SELECT count(*) FILTER (WHERE ttft_ms IS NULL) null_ttft, count(*) total
FROM trace_summary WHERE created_at::timestamptz>=now()-interval '1 day' AND workflow_name='agent';
```
